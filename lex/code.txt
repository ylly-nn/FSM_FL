{
    a, b, i : %;
    x, y : !;
    flag, ok : $;
    

    readln a, b;

    x := a div 2;
    y := (a mult b) plus 3.5;

    ok := (a LT b);

    if (flag) begin
        writeln x, y;
        flag := false
    end else begin
        writeln 0;
        flag := true
    end;

    for i := 0 to 10 step 2 begin
        a := a plus i;
        writeln a
    end 

    while (true) a:=b;
}
