
{    a, b, c : %;
    x : !;
    flag : $;
    i : %;

    readln a, b;

    c := (a mult b) plus 10;
    writeln c;

    flag := true;

    if ((c GT 100) and flag) begin
        writeln c;
        x := ~a
    end else begin
        writeln 0;
        flag := false
    end;

    for i := 0 to 10 step 2 begin
        a := a plus i;
        writeln a
    end next;

    while (a LT b) begin
        a := a plus 1
    end;

    writeln (a EQ b);
    writeln (a NE b);
    writeln (a LE b);
    writeln (a GE b);
    
}
